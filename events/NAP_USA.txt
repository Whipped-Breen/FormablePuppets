add_namespace = NAP_american_surrender

#The Tiergartenstraße Conference
country_event = {
	id = NAP_american_surrender.1
	title = NAP_american_surrender.1.t
	desc = NAP_american_surrender.1.d
	picture = GFX_report_event_generic_sign_treaty2

	fire_only_once = yes

	trigger = {
		#A democratic faction leader whose faction holds at least one German core state while SOV holds another can get this.
		has_war_with = GER
		USA = {
			surrender_progress > 0.4
			has_government = democratic
		}
		is_faction_leader = yes
		any_state = {
			is_core_of = GER
			controller = {
				is_in_faction_with = ROOT
			}
		}
		has_government = fascism
		NOT = {
			tag = JAP
		}
		NOT = {
			has_war_with = JAP
		}
		JAP = {
			is_faction_leader = yes
			has_war_with = USA
		}
		NOT = {
			has_global_flag = had_tiergartenstrasze_conference
		}
		NOT = {
			any_state = {
				is_core_of = USA
				controller = {
					NOT = {
						tag = USA
					}
					is_in_faction = yes
					NOT = {
						OR = {
							is_in_faction_with = ROOT
							is_in_faction_with = JAP
						}
					}
				}
			}
		}
	}

	mean_time_to_happen = {
		days = 8
	}

	immediate = {
		set_global_flag = had_tiergartenstrasze_conference
		JAP = {
			save_event_target_as = tiergartenstrasze_partner
		}
		ROOT = {
			save_event_target_as = tiergartenstrasze_origin
			set_country_flag = tiergartenstrasze_signatory_faction_leader
		}
	}

	option = {
		name = NAP_american_surrender.1.a
		ai_chance = {
			factor = 10
		}
		hidden_effect = {
			event_target:tiergartenstrasze_partner = {
				country_event = { days = 1 id = NAP_american_surrender.2 }
			}
		}
	#	custom_effect_tooltip = tiergartenstrasze_division_tt
		event_target:tiergartenstrasze_partner = {
			custom_effect_tooltip = needs_approval
		}
	}
}
